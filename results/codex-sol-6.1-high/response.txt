      function mesh(parent, geometry, mat, position, scale) {
        const object = new THREE.Mesh(geometry, mat);
        object.position.set(...position);
        if (scale) object.scale.set(...scale);
        object.castShadow = true;
        object.receiveShadow = true;
        parent.add(object);
        return object;
      }

      function ellipsoid(parent, mat, position, scale, coarse = false) {
        return mesh(parent, coarse ? coarseGeometry : sphereGeometry, mat, position, scale);
      }

      function bone(parent, start, end, topRadius, bottomRadius, mat, sides = 6) {
        const a = new THREE.Vector3(...start);
        const b = new THREE.Vector3(...end);
        const direction = b.clone().sub(a);
        const geometry = new THREE.CylinderGeometry(bottomRadius, topRadius, direction.length(), sides);
        const object = mesh(parent, geometry, mat, a.clone().add(b).multiplyScalar(0.5).toArray());
        object.quaternion.setFromUnitVectors(up, direction.normalize());
        return object;
      }

          updatePlaybackUI();

      let pointer = null;
      renderer.domElement.addEventListener('pointerdown', event => {
        if (pointer || (event.pointerType === 'mouse' && event.button !== 0)) return;

        renderer.domElement.setPointerCapture(event.pointerId);
      });

        if (!pointer || pointer.id !== event.pointerId) return;

          desiredView.pitch + (event.clientY - pointer.y) * 0.005, 0.05, 0.75
        );
        pointer.x = event.clientX;
        pointer.y = event.clientY;
      });

        if (pointer && pointer.id === event.pointerId) pointer = null;

      function resize() {
        const width = stage.clientWidth;
        const height = stage.clientHeight;
        renderer.setSize(width, height);
        camera.aspect = width / height;
        camera.fov = camera.aspect < 1 ? 43 : 34;
        camera.updateProjectionMatrix();
      }
      new ResizeObserver(resize).observe(stage);
      resize();

      const hipWorld = new THREE.Vector3();
      const footWorld = new THREE.Vector3();
      const footLocal = new THREE.Vector3();
      const smoothstep = x => x * x * (3 - 2 * x);
      const wrap = (value, length) => ((value % length) + length) % length;

      function animateHorse(time) {
        const cycle = wrap(time / cycleDuration, 1);
        const phase = cycle * tau;
        horse.position.y = 1.98 + Math.sin(phase - 0.7) * 0.07;
        horse.rotation.z = Math.sin(phase + 0.5) * 0.035;
        horse.rotation.x = Math.sin(phase) * 0.013;
        neck.rotation.z = Math.sin(phase - 0.65) * 0.045;
        head.rotation.z = Math.sin(phase - 1.0) * 0.035;
        tail.rotation.y = Math.sin(phase - 0.5) * 0.20;
        tail.rotation.z = 0.12 + Math.sin(phase + 0.4) * 0.10;
        tailSegments.forEach((segment, index) => {
          segment.rotation.z = Math.sin(phase - index * 0.48) * 0.09;
          segment.rotation.y = Math.sin(phase - index * 0.55) * 0.07;
        });
        horse.updateMatrixWorld(true);

        for (const leg of legs) {
          const p = wrap(cycle + leg.phase, 1);
          const stance = 0.36;
          let stride;
          let lift;
          let footPitch;
          if (p < stance) {
            const f = p / stance;
            stride = 0.54 - f * 1.08;
            lift = 0;
            footPitch = 0;
          } else {
            const f = (p - stance) / (1 - stance);
            stride = -0.54 + smoothstep(f) * 1.08;
            lift = Math.pow(Math.sin(Math.PI * f), 1.5) * (leg.front ? 0.65 : 0.55);
            footPitch = Math.sin(Math.PI * f) * (leg.front ? -0.45 : 0.3);
          }
          leg.root.getWorldPosition(hipWorld);
          footWorld.set(hipWorld.x + stride, 0.155 + lift, hipWorld.z);
          footLocal.copy(footWorld);
          horse.worldToLocal(footLocal);
          const dx = footLocal.x - leg.root.position.x;
          const dy = footLocal.y - leg.root.position.y;
          const distance = Math.min(Math.hypot(dx, dy), (upperLength + lowerLength) * 0.998);
          const bend = Math.acos(THREE.MathUtils.clamp(
            (distance * distance - upperLength * upperLength - lowerLength * lowerLength) /
            (2 * upperLength * lowerLength), -1, 1
          )) * (leg.front ? -1 : 1);
          const angle = Math.atan2(dx, -dy) - Math.atan2(
            lowerLength * Math.sin(bend), upperLength + lowerLength * Math.cos(bend)
          );
          leg.upper.rotation.z = angle;
          leg.lower.rotation.z = bend;
          leg.hoof.rotation.z = -angle - bend - horse.rotation.z + footPitch;
        }

        contactShadow.material.opacity = 0.73 - Math.sin(phase - 0.7) * 0.07;
        for (const item of scenery) {
          item.group.position.x = wrap(item.initialX - time * worldSpeed + terrainLength / 2, terrainLength) - terrainLength / 2;
        }
        for (const item of dust) {
          const p = wrap(cycle + item.offset, 1);
          item.particle.position.set(
            -1.05 - p * 3.1,
            0.09 + Math.sin(Math.PI * p) * item.rise,
            item.spread * (0.5 + p)
          );
          item.particle.scale.setScalar(item.size * (0.55 + p * 1.7));
          item.particle.material.opacity = 0.25 * Math.pow(1 - p, 2) * Math.min(p * 12, 1);
        }
      }

      document.addEventListener('visibilitychange', () => { lastTime = null; });

      function frame(now) {
        requestAnimationFrame(frame);
        const delta = lastTime === null ? 0 : Math.min((now - lastTime) / 1000, 0.05);
        lastTime = now;
        if (playing) elapsed = wrap(elapsed + delta * speed, cycleDuration * 8);
        const easing = 1 - Math.exp(-delta * 9);
        view.yaw += (desiredView.yaw - view.yaw) * easing;
        view.pitch += (desiredView.pitch - view.pitch) * easing;
        view.distance += (desiredView.distance - view.distance) * easing;
        const horizontal = Math.cos(view.pitch) * view.distance;
        camera.position.set(
          target.x + Math.sin(view.yaw) * horizontal,
          target.y + Math.sin(view.pitch) * view.distance,
          target.z + Math.cos(view.yaw) * horizontal
        );
        camera.lookAt(target);
        animateHorse(elapsed);
        renderer.render(scene, camera);
      }
      requestAnimationFrame(frame);

      document.getElementById('live-dot').style.background = '#ac512e';

    }
